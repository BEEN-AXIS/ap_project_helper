SELECT id, mode, map, status, created_at, end_reason, winner_side, player1_name, player2_name
FROM (SELECT *
      FROM matches
      ORDER BY created_at DESC
      LIMIT 100) recent
WHERE lower(player1_name || player2_name || id) LIKE '%player_1%'
ORDER BY created_at DESC;
